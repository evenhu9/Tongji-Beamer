\section{研究背景}

\begin{frame}{研究背景与问题}
    \scriptsize
    \begin{block}{安全形势与研究现状}
        \begin{itemize}
            \item 渗透测试是Web安全防护的关键，但传统人工方式存在效率低、成本高、依赖专家经验等明显缺陷
            \item 大语言模型（LLM）的发展使利用LLM Agent 实现自动化渗透测试成为当前网络安全领域的研究热点之一
            \item AutoPT\cite{Wu2024AutoPT} 利用\textbf{渗透测试状态机（PSM）}来引导LLM Agent 执行渗透测试任务
    \end{block}

    \begin{alertblock}{原始 PSM 存在缺陷}
        \begin{enumerate}
            \item \textbf{PSM 流程控制}：状态粒度过粗、转移条件固定、提示词缺乏角色定义
            \item \textbf{工具体系缺口}：模拟浏览器交互能力弱、ReAct 解析器无法处理思维链输出、poc提取工具误差大
            \item \textbf{系统可用性不足}：仅命令行，无法直观监控进度
        \end{enumerate}
    \end{alertblock}

    \begin{exampleblock}{核心目标}
        (1) PSM 流程控制优化 $\quad$ (2) 工具扩展与容错增强 $\quad$ (3) 图形化界面设计
    \end{exampleblock}
\end{frame}
