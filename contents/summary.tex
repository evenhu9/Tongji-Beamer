\section{总结与展望}

\begin{frame}{工作总结与展望}
    \footnotesize
    \begin{columns}
        \column{.52\textwidth}
        \begin{block}{三大改进与实验结论}
            \begin{itemize}
                \item \textbf{PSM 流程控制}：Scan 四子阶段 + 提前终止 + 上下文压缩 + Check 双层验证
                \item \textbf{工具与容错}：三个浏览器扩展工具 + RobustReActParser 多级降级
                \item \textbf{Web Console}：六模块图形化管理界面
            \end{itemize}
            \vspace{0.5em}
            基于 300 次独立实验：总体成功率 24.67\% $\to$ \textbf{63.67\%（+39pp）}；成本 $-$69\%，耗时 $-$42\%；工具执行失败仅 \textbf{10.1\%}
        \end{block}

        \column{.44\textwidth}
        \begin{alertblock}{现有不足}
            \begin{enumerate}
                \item \textbf{提前放弃（33\%）}：重试建议不区分失败类型
                \item \textbf{安全审查拦截（22\%）}：模型拒绝生成攻击性命令
                \item \textbf{覆盖范围局限}：仅 Web 应用层 CVE
            \end{enumerate}
        \end{alertblock}
        \begin{block}{未来工作}
            \begin{itemize}
                \item 精细化重试策略（按失败类型）
                \item 引入本地开源模型（如 DeepSeek）
                \item 扩展 PSM 覆盖完整渗透测试阶段
            \end{itemize}
        \end{block}
    \end{columns}
\end{frame}
