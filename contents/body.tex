\begin{frame}{PSM 渗透测试状态机}
    \scriptsize
    \begin{columns}
        \column{.45\textwidth}
        \begin{figure}
            \centering
            \includegraphics[width=\columnwidth,height=0.6\textheight,keepaspectratio]{contents/figure/psm-original.png}
            \caption{原版 PSM 模型\cite{Wu2024AutoPT}}
        \end{figure}

        \column{.52\textwidth}
        \begin{block}{状态定义}
            \begin{enumerate}
                \item \textbf{Scan}：端口扫描 \& 服务识别
                \item \textbf{VulnSelect}：筛选值得尝试的漏洞
                \item \textbf{Inquire}：检索外部 PoC 与利用方法
                \item \textbf{Exploit}：执行漏洞利用操作
                \item \textbf{Check}：验证利用是否成功，失败则回退至 Exploit 或 VulnSelect
            \end{enumerate}
        \end{block}
        \begin{block}{实现}
            基于 LangGraph StateGraph实现，状态为图中节点，路由函数根据状态输出决定转移路径
        \end{block}
    \end{columns}
\end{frame}

\section{改进一：PSM 流程控制}

\begin{frame}{改进一：PSM 流程控制优化}
    \scriptsize
    \begin{columns}
        \column{.48\textwidth}
        \begin{figure}
            \centering
            \includegraphics[width=\columnwidth,height=0.80\textheight,keepaspectratio]{contents/figure/psm.png}
            \caption{改进前后状态转移模型对比}
        \end{figure}

        \column{.52\textwidth}
        \begin{itemize}
            \item \textbf{提示词规范}：统一 ReAct 模板 + 差异化角色
            \item \textbf{提前终止}：parse\_vuln 判空跳过下游（节省 $\sim$200s）
            \item \textbf{Scan 细化}：nmap$\to$services.yml$\to$xray$\to$评估；Exploit 7.2$\to$4.8
            \item \textbf{PoC + Check}：容错机制；匹配域缩小 + LLM+正则双层校验
            \item \textbf{上下文压缩}：即时摘要 + 结构化组装
        \end{itemize}
    \end{columns}
\end{frame}

\section{实验与评估}

\begin{frame}{实验与评估}
    \scriptsize
    \begin{block}{实验设置}
        20 个 CVE（10 简单 + 10 复杂）$\times$ 3 模型 $\times$ 5 次 = \textbf{300 组独立实验}
    \end{block}

    \begin{columns}
        \column{.48\textwidth}
        \begin{exampleblock}{RQ1：模型差异}
            \textbf{63.67\%（+39pp）} $\;|\;$ 4o-mini 74\% 最优 $\;|\;$ 3.5 $\sim$7\%$\to$54\% 增幅最大
        \end{exampleblock}

        \column{.48\textwidth}
        \begin{exampleblock}{RQ2：成本效益}
            费用 $-$69\% $\;|\;$ 耗时 $-$42\% $\;|\;$ 单次 \$0.003109 / 94.2s
        \end{exampleblock}
    \end{columns}

    \vspace{0.3em}
    \begin{exampleblock}{核心结论}
        成功率 24.67\%$\to$\textbf{63.67\%} $\;|\;$ 原版 38/60 组 0\% $\to$ 仅 \textbf{3 组}
    \end{exampleblock}
\end{frame}

\begin{frame}{RQ3：改进效果评估}
    \scriptsize
    \begin{columns}
        \column{.52\textwidth}
        \begin{block}{改进效果}
            \textbf{PSM 流程}：提前终止省去无效执行，上下文压缩控制 Token 增长，提示词规范化减少越权命令 \\[0.3em]
            \textbf{工具扩展}：浏览器 + 容错解析 + PoC 增强，工具执行失败仅 \textbf{10.1\%}，错误的命令 29.4\%（4o-mini 最低 26.92\%） \\[0.3em]
            \textbf{Web Console}：提高实验过程一致性与结果可复核性，保障 300 组实验基础支撑
