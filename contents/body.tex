\section{系统设计}

\begin{frame}{PSM 渗透测试状态机}
    \scriptsize
    \begin{columns}
        \column{.48\textwidth}
        \begin{figure}
            \centering
            \includegraphics[width=\columnwidth,height=0.70\textheight,keepaspectratio]{contents/figure/psm-original.png}
            \caption{原版 PSM 模型\cite{Wu2024AutoPT}}
        \end{figure}

        \column{.52\textwidth}
        \begin{block}{状态定义}
            \begin{enumerate}
                \item \textbf{Scan}：执行端口与漏洞扫描
                \item \textbf{VulnSelect}：从扫描结果中选择漏洞
                \item \textbf{Inquire}：针对选定漏洞，检索外部 PoC 与利用方法
                \item \textbf{Exploit}：调用工具执行漏洞利用操作，使用playwright模拟浏览器交互
                \item \textbf{Check}：验证利用是否成功，失败则回退至 Exploit 或 VulnSelect
            \end{enumerate}
        \end{block}
        \begin{block}{实现}
            基于 LangGraph 实现，状态为图中节点，路由函数根据状态输出决定转移路径
        \end{block}
    \end{columns}
\end{frame}

\begin{frame}{系统总体架构}
    \scriptsize
    \begin{columns}
        \column{.40\textwidth}
        \begin{figure}
            \centering
            \includegraphics[width=\columnwidth,height=0.55\textheight,keepaspectratio]{contents/figure/arch.png}
            \caption{AutoPT+ 架构图}
        \end{figure}

        \column{.46\textwidth}
        \begin{block}{分层设计}
            \begin{itemize}
                \item \textbf{表示层}：Web Console 提供图形化管理界面，管理docker靶机、渗透测试状态与实验数据分析
                \item \textbf{业务逻辑层}：基于 Flask API 构建，负责处理请求与任务调度
                \item \textbf{核心引擎层}：基于LangGraph 实现的 PSM 引擎，是渗透测试的实际执行者
                \item \textbf{基础设施层}：集成 Nmap、Xray、Playwright 等专业工具，提供底层的扫描与验证能力
            \end{itemize}
        \end{block}
    \end{columns}
\end{frame}

\section{改进一：PSM 流程控制}

\begin{frame}{改进一：PSM 流程控制优化}
    \scriptsize
    \begin{columns}
        \column{.48\textwidth}
        \begin{figure}
            \centering
            \includegraphics[width=\columnwidth,height=0.70\textheight,keepaspectratio]{contents/figure/psm.png}
            \caption{改进前后状态转移模型对比}
        \end{figure}

        \column{.52\textwidth}
        \begin{itemize}
            \item \textbf{提前终止机制}：新增条件判断，若漏洞列表为空，则跳过下游状态直接结束
            \item \textbf{Scan子阶段划分}：将原单一阶段拆分为：端口发现与服务识别 → 服务知识匹配 → 漏洞扫描 → 结果评估与终止判断
            \item \textbf{PoC容错提取}：Inquire Agent尝试获取PoC时依次从 Agent 的 Final Answer、推理文本、ReadHTML 原始捕获内容和原始输出四个层级尝试提取，避免因解析失败而丢失数据
            \item \textbf{Check结果匹配}：check阶段只对工具输出做目标结果匹配，并引入 LLM 自主判断是否成功
            \item \textbf{提示词规范}：提示词加入统一的格式约束，并对不同的agent设计不同的角色定义和权限边界
            \item \textbf{上下文压缩}：将LLM上下文按照完整保留任务目标、对历史对话做LLM摘要、截取最近两条工具输出的流程进行压缩
        \end{itemize}
    \end{columns}
\end{frame}

\section{改进三：Web Console}

\begin{frame}{Web Console 图形化管理界面}
    \scriptsize
    \begin{columns}
        \column{.45\textwidth}
        \begin{block}{功能模块}
            \begin{itemize}
                \item \textbf{控制台}：总览系统状态与web console所有功能入口
                \item \textbf{漏洞库}：展示基准测试的20个CVE ，支持关键字搜索、漏洞分类/难度筛选
                \item \textbf{渗透测试}：能够一键创建任务，支持langgraph的debug模式，实时显示测试进度与状态
                \item \textbf{靶机管理}：支持一键对Docker 容器启/停/销毁，并展示运行中的容器列表
                \item \textbf{测试报告}：显示测试历史记录，从成功率、成本、工具调用次数等多个维度进行分析
                \item \textbf{系统设置}：可配置LLM模型参数和PSM流程控制参数，同时显示系统信息
            \end{itemize}
        \end{block}

\section{实验与评估}

\begin{frame}{实验与评估}
    \scriptsize
    \begin{block}{实验设置}
        20 个 CVE（10 简单 + 10 复杂）$\times$ 3 模型 $\times$ 5 次 = \textbf{300 组独立实验}
    \end{block}

    \begin{columns}
        \column{.48\textwidth}
        \begin{exampleblock}{RQ1：模型差异}
            \textbf{63.67\%（+39pp）} $\;|\;$ 4o-mini 74\% 最优 $\;|\;$ 3.5 $\sim$7\%$\to$54\% 增幅最大
        \end{exampleblock}

        \column{.48\textwidth}
        \begin{exampleblock}{RQ2：成本效益}
            费用 $-$69\% $\;|\;$ 耗时 $-$42\% $\;|\;$ 单次 \$0.003109 / 94.2s
        \end{exampleblock}
    \end{columns}

    \vspace{0.3em}
    \begin{exampleblock}{核心结论}
        成功率 24.67\%$\to$\textbf{63.67\%} $\;|\;$ 原版 38/60 组 0\% $\to$ 仅 \textbf{3 组}
    \end{exampleblock}
\end{frame}

\begin{frame}{RQ3：改进效果评估}
    \scriptsize
